```python
# scan_en.py — hardcoded English JSX text nodes
import glob, re
from collections import Counter
pat = re.compile(r">([A-Z][A-Za-z0-9 .,!?:;()\-/']{3,})<")
hits = []
for root in ["components", "app", "context"]:
    for f in glob.glob(f"{root}/**/*.tsx", recursive=True):
        if "node_modules" in f: continue
        src = open(f, encoding="utf-8").read()
        for m in pat.finditer(src):
            txt = m.group(1).strip()
            if txt and not txt.endswith("="): hits.append((f, txt))
print("JSX English text-node hits:", len(hits))
for f, t in hits[:20]: print(f"  {t!r}  ({f})")
```
